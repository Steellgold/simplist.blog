# Update Page View

Update engagement metrics for an existing page view, typically called when the user leaves the page.

```http
PUT /v1/analytics/track/:pageViewId
```

<Callout type="info">
Use this endpoint to record final engagement metrics like time on page and scroll depth before the user navigates away.
</Callout>

## Path Parameters

| Parameter | Type | Required | Description |
|-----------|------|----------|-------------|
| `pageViewId` | string | Yes | Page view ID from initial track call |

## Request Body

| Field | Type | Description |
|-------|------|-------------|
| `timeOnPage` | number | Total time spent on page (seconds) |
| `scrollDepth` | number | Maximum scroll depth percentage (0-100) |
| `exitPosition` | number | Scroll position when exiting |
| `bounced` | boolean | Whether the user bounced (left without interaction) |

## Complete Tracking Flow

Here's a complete example tracking a page view from start to finish:

```typescript
class PageAnalytics {
  private pageViewId: string | null = null;
  private startTime: number = Date.now();
  private maxScrollDepth: number = 0;

  async trackPageLoad(articleSlug: string) {
    // Initial page view
    const response = await fetch('https://api.simplist.blog/v1/analytics/track', {
      method: 'POST',
      headers: {
        'X-API-Key': process.env.NEXT_PUBLIC_SIMPLIST_API_KEY!,
        'Content-Type': 'application/json'
      },
      body: JSON.stringify({
        articleSlug,
        visitorId: this.getVisitorId(),
        sessionId: this.getSessionId(),
        pageUrl: window.location.href,
        referrer: document.referrer,
        screenWidth: window.screen.width,
        screenHeight: window.screen.height
      })
    });

    const { pageViewId } = await response.json();
    this.pageViewId = pageViewId;

    // Track scroll depth
    window.addEventListener('scroll', () => this.updateScrollDepth());

    // Update metrics before page unload
    window.addEventListener('beforeunload', () => this.trackPageExit());
  }

  private updateScrollDepth() {
    const scrollPercent = Math.min(
      100,
      Math.round(
        ((window.scrollY + window.innerHeight) / document.body.scrollHeight) * 100
      )
    );

    this.maxScrollDepth = Math.max(this.maxScrollDepth, scrollPercent);
  }

  private async trackPageExit() {
    if (!this.pageViewId) return;

    const timeOnPage = Math.floor((Date.now() - this.startTime) / 1000);
    const bounced = this.maxScrollDepth < 25 && timeOnPage < 10;

    // Use sendBeacon for reliable tracking on page unload
    navigator.sendBeacon(
      `https://api.simplist.blog/v1/analytics/track/${this.pageViewId}`,
      JSON.stringify({
        timeOnPage,
        scrollDepth: this.maxScrollDepth,
        exitPosition: window.scrollY,
        bounced
      })
    );
  }

  private getVisitorId(): string {
    let visitorId = localStorage.getItem('visitorId');
    if (!visitorId) {
      visitorId = `visitor_${Date.now()}_${crypto.randomUUID()}`;
      localStorage.setItem('visitorId', visitorId);
    }
    return visitorId;
  }

  private getSessionId(): string {
    let sessionId = sessionStorage.getItem('sessionId');
    if (!sessionId) {
      sessionId = `session_${Date.now()}_${crypto.randomUUID()}`;
      sessionStorage.setItem('sessionId', sessionId);
    }
    return sessionId;
  }
}

// Usage
const analytics = new PageAnalytics();
analytics.trackPageLoad('getting-started');
```

## Using Navigator.sendBeacon

For reliable tracking on page unload, use `navigator.sendBeacon()`:

```javascript
window.addEventListener('beforeunload', () => {
  const pageViewId = sessionStorage.getItem('pageViewId');
  if (!pageViewId) return;

  const data = JSON.stringify({
    timeOnPage: Math.floor((Date.now() - pageLoadTime) / 1000),
    scrollDepth: maxScrollDepth,
    exitPosition: window.scrollY,
    bounced: false
  });

  navigator.sendBeacon(
    `https://api.simplist.blog/v1/analytics/track/${pageViewId}`,
    new Blob([data], { type: 'application/json' })
  );
});
```

<Callout type="info">
`navigator.sendBeacon()` ensures the request completes even if the page is unloading, unlike regular `fetch()`.
</Callout>

## Calculating Bounce Rate

A bounce typically means the user left without meaningful engagement:

```typescript
function didUserBounce(timeOnPage: number, scrollDepth: number): boolean {
  // Bounced if spent less than 10 seconds AND scrolled less than 25%
  return timeOnPage < 10 && scrollDepth < 25;
}

const bounced = didUserBounce(timeOnPage, maxScrollDepth);
```

## Best Practices

### Track Scroll Depth Efficiently

Throttle scroll events to avoid excessive updates:

```typescript
let maxScrollDepth = 0;
let scrollTimeout: NodeJS.Timeout;

window.addEventListener('scroll', () => {
  clearTimeout(scrollTimeout);

  scrollTimeout = setTimeout(() => {
    const scrollPercent = Math.min(
      100,
      Math.round(
        ((window.scrollY + window.innerHeight) / document.body.scrollHeight) * 100
      )
    );

    maxScrollDepth = Math.max(maxScrollDepth, scrollPercent);
  }, 100); // Update every 100ms
});
```

### Calculate Time on Page Accurately

```typescript
const pageLoadTime = Date.now();

function getTimeOnPage(): number {
  return Math.floor((Date.now() - pageLoadTime) / 1000);
}
```

### Handle Visibility Changes

Pause time tracking when page is hidden:

```typescript
let totalTimeOnPage = 0;
let lastVisibleTime = Date.now();
let isPageVisible = !document.hidden;

document.addEventListener('visibilitychange', () => {
  if (document.hidden) {
    // Page hidden - add elapsed time
    if (isPageVisible) {
      totalTimeOnPage += (Date.now() - lastVisibleTime) / 1000;
      isPageVisible = false;
    }
  } else {
    // Page visible again
    lastVisibleTime = Date.now();
    isPageVisible = true;
  }
});

function getTotalTimeOnPage(): number {
  if (isPageVisible) {
    return totalTimeOnPage + (Date.now() - lastVisibleTime) / 1000;
  }
  return totalTimeOnPage;
}
```
