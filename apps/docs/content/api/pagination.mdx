# Pagination

List endpoints in the Simplist API return paginated results to ensure fast response times and efficient data transfer. All paginated responses follow a consistent structure.

## Pagination Parameters

Control pagination using query parameters:

| Parameter | Type | Default | Description |
|-----------|------|---------|-------------|
| `page` | integer | `1` | Page number to retrieve (1-indexed) |
| `limit` | integer | `10` | Number of items per page (max: 100) |

## Example Request

```bash
curl "https://api.simplist.blog/v1/articles?page=2&limit=20" \
  -H "X-API-Key: your_api_key"
```

## Response Structure

All paginated responses include `data` and `meta` fields:

```json
{
  "data": [
    {
      "id": "clx123",
      "title": "First Article",
      "slug": "first-article"
    },
    {
      "id": "clx124",
      "title": "Second Article",
      "slug": "second-article"
    }
  ],
  "meta": {
    "total": 47,
    "page": 2,
    "limit": 20,
    "totalPages": 3
  }
}
```

### Response Fields

#### `data`

Array of resource objects for the current page.

#### `meta`

Pagination metadata:

| Field | Type | Description |
|-------|------|-------------|
| `total` | integer | Total number of items across all pages |
| `page` | integer | Current page number |
| `limit` | integer | Items per page |
| `totalPages` | integer | Total number of pages |

## Implementation Examples

### Fetching All Pages

```typescript
async function fetchAllArticles(apiKey: string) {
  const allArticles = [];
  let page = 1;
  let hasMore = true;

  while (hasMore) {
    const response = await fetch(
      `https://api.simplist.blog/v1/articles?page=${page}&limit=100`,
      { headers: { 'X-API-Key': apiKey } }
    );

    const { data, meta } = await response.json();

    allArticles.push(...data);

    hasMore = page < meta.totalPages;
    page++;
  }

  return allArticles;
}
```

### Pagination with Async Iteration

```typescript
async function* paginateArticles(apiKey: string, limit = 50) {
  let page = 1;
  let hasMore = true;

  while (hasMore) {
    const response = await fetch(
      `https://api.simplist.blog/v1/articles?page=${page}&limit=${limit}`,
      { headers: { 'X-API-Key': apiKey } }
    );

    const { data, meta } = await response.json();

    yield data;

    hasMore = page < meta.totalPages;
    page++;
  }
}

// Usage
for await (const articles of paginateArticles(apiKey)) {
  console.log(`Processing ${articles.length} articles...`);
  // Process each page
}
```

### React Hook for Pagination

```typescript
import { useState, useEffect } from 'react';

interface PaginatedResponse<T> {
  data: T[];
  meta: {
    total: number;
    page: number;
    limit: number;
    totalPages: number;
  };
}

function usePaginatedArticles(apiKey: string, page: number, limit: number) {
  const [data, setData] = useState<PaginatedResponse<Article> | null>(null);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState<Error | null>(null);

  useEffect(() => {
    async function fetchData() {
      try {
        setLoading(true);
        const response = await fetch(
          `https://api.simplist.blog/v1/articles?page=${page}&limit=${limit}`,
          { headers: { 'X-API-Key': apiKey } }
        );

        if (!response.ok) throw new Error('Failed to fetch');

        const result = await response.json();
        setData(result);
      } catch (err) {
        setError(err instanceof Error ? err : new Error('Unknown error'));
      } finally {
        setLoading(false);
      }
    }

    fetchData();
  }, [apiKey, page, limit]);

  return { data, loading, error };
}

// Usage
function ArticleList() {
  const [page, setPage] = useState(1);
  const { data, loading, error } = usePaginatedArticles(apiKey, page, 20);

  if (loading) return <div>Loading...</div>;
  if (error) return <div>Error: {error.message}</div>;

  return (
    <>
      {data?.data.map(article => (
        <ArticleCard key={article.id} article={article} />
      ))}

      <Pagination
        currentPage={data?.meta.page || 1}
        totalPages={data?.meta.totalPages || 1}
        onPageChange={setPage}
      />
    </>
  );
}
```

## Best Practices

### 1. Use Reasonable Page Sizes

Balance between API calls and response size:

```typescript
// Good: Reasonable page size
const limit = 50; // 50 items per page

// Avoid: Too small (many API calls)
const limit = 5;

// Avoid: Too large (slow responses)
const limit = 1000; // Max is 100 anyway
```

### 2. Cache Paginated Results

Avoid refetching the same pages:

```typescript
const cache = new Map<number, Article[]>();

async function getCachedPage(page: number, limit: number, apiKey: string) {
  const cacheKey = page;

  if (cache.has(cacheKey)) {
    return cache.get(cacheKey);
  }

  const response = await fetch(
    `https://api.simplist.blog/v1/articles?page=${page}&limit=${limit}`,
    { headers: { 'X-API-Key': apiKey } }
  );

  const { data } = await response.json();
  cache.set(cacheKey, data);

  return data;
}
```

### 3. Handle Empty Pages

```typescript
const { data, meta } = await response.json();

if (data.length === 0) {
  if (meta.total === 0) {
    console.log('No articles found');
  } else {
    console.log('Page is empty (past last page)');
  }
}
```

### 4. Validate Page Numbers

```typescript
function validatePageNumber(page: number, totalPages: number): number {
  // Ensure page is at least 1
  if (page < 1) return 1;

  // Ensure page doesn't exceed total pages
  if (page > totalPages) return totalPages;

  return page;
}
```

## Limits

<Callout type="info">
The maximum `limit` value is **100 items per page**. Requests with higher limits will be capped at 100.
</Callout>

```bash
# This will return max 100 items, not 500
curl "https://api.simplist.blog/v1/articles?limit=500" \
  -H "X-API-Key: your_api_key"
```
