# /v1/tags

Retrieve all tags from your Simplist project with article counts and metadata.

## Endpoint

<ApiPath
  method="GET"
  path="/v1/tags"
  baseUrl="https://api.simplist.blog"
  testable={true}
  requiresAuth={true}
/>

## Query Parameters

<TypeTable
  title="Query Parameters"
  properties={[
    {
      name: "sort",
      type: "'name' | 'createdAt' | 'articleCount'",
      description: "Field to sort by",
      defaultValue: "name",
    },
    {
      name: "order",
      type: "'asc' | 'desc'",
      description: "Sort direction",
      defaultValue: "asc",
    },
    {
      name: "limit",
      type: "number",
      description: "Maximum number of tags to return",
    },
  ]}
/>

## Response

Returns an array of `TagListItem` objects with metadata about total count.

<TypeTable
  title="TagListItem"
  properties={[
    {
      name: "id",
      type: "string",
      description: "Unique tag identifier",
      required: true,
    },
    { name: "name", type: "string", description: "Tag name", required: true },
    {
      name: "icon",
      type: "string | null",
      description: "Tag icon name (e.g., 'code', 'book')",
    },
    {
      name: "color",
      type: "string | null",
      description: "Hex color code (e.g., '#3B82F6')",
    },
    {
      name: "articleCount",
      type: "number",
      description: "Number of articles with this tag",
      required: true,
    },
    {
      name: "createdAt",
      type: "string",
      description: "ISO 8601 creation timestamp",
      required: true,
    },
    {
      name: "updatedAt",
      type: "string",
      description: "ISO 8601 last update timestamp",
      required: true,
    },
  ]}
/>

## Example Request

<CurlCommand
  method="GET"
  url="https://api.simplist.blog/v1/tags"
  headers={{
    "X-API-Key": "YOUR_API_KEY",
  }}
/>

### With Sorting and Limit

<CurlCommand
  method="GET"
  url="https://api.simplist.blog/v1/tags?sort=articleCount&order=desc&limit=5"
  headers={{
    "X-API-Key": "YOUR_API_KEY",
  }}
/>
