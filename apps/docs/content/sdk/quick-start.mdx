# Quick Start

Get up and running with the Simplist SDK in just a few minutes.

## Installation

<Steps>
<Step>
### Install the package

<Tabs items={['npm', 'pnpm', 'yarn']}>
<Tab value="npm">
```bash
npm install @simplist.blog/sdk
```
</Tab>
<Tab value="pnpm">
```bash
pnpm add @simplist.blog/sdk
```
</Tab>
<Tab value="yarn">
```bash
yarn add @simplist.blog/sdk
```
</Tab>
</Tabs>
</Step>

<Step>
### Get your API key

1. Go to your [project settings](https://app.simplist.blog)
2. Navigate to **API Keys**
3. Create a new API key
4. Copy the key (starts with `prj_` or `pk_`)
</Step>

<Step>
### Set up environment variable

```bash title=".env"
SIMPLIST_API_KEY=your_api_key
```

<Callout type="warn">
Never commit API keys to version control. Always use environment variables.
</Callout>
</Step>
</Steps>

## Your First API Call

```typescript
import { SimplistClient } from '@simplist.blog/sdk';

// Create client (auto-detects SIMPLIST_API_KEY from env)
const client = new SimplistClient();

// Fetch your articles
const { data: articles, meta } = await client.articles.list({
  published: true,
  limit: 10,
});

console.log(`Found ${meta.total} articles`);
articles.forEach(article => {
  console.log(`- ${article.title}`);
});
```

## Complete Example

Here's a complete example showing common operations:

```typescript
import { SimplistClient } from '@simplist.blog/sdk';

const client = new SimplistClient({
  apiKey: process.env.SIMPLIST_API_KEY,
});

async function main() {
  // 1. Get project info
  const { data: project } = await client.project.get();
  console.log(`Project: ${project.project.name}`);
  console.log(`Articles: ${project.stats.totalArticles}`);
  console.log(`Views: ${project.stats.totalViews}`);

  // 2. List articles
  const { data: articles } = await client.articles.list({
    published: true,
    limit: 5,
    sort: 'publishedAt',
    order: 'desc',
  });

  console.log('\nLatest Articles:');
  articles.forEach(article => {
    console.log(`- ${article.title} (${article.viewCount} views)`);
  });

  // 3. Get specific article
  const { data: article } = await client.articles.get('getting-started');
  console.log(`\nArticle: ${article.title}`);
  console.log(`Reading time: ${article.readTimeMinutes} minutes`);

  // 4. Track analytics (if using public key in browser)
  await client.analytics.track({
    articleSlug: 'getting-started',
    visitorId: 'visitor_123',
    sessionId: 'session_456',
    pageUrl: 'https://myblog.com/articles/getting-started',
    referrer: 'https://google.com',
  });

  console.log('✓ Page view tracked');

  // 5. Get analytics stats (requires read permission)
  const stats = await client.analytics.getStats({ days: 30 });
  console.log(`\nAnalytics (last 30 days):`);
  console.log(`- Total views: ${stats.summary.totalViews}`);
  console.log(`- Unique visitors: ${stats.summary.uniqueVisitors}`);
}

main().catch(console.error);
```

## Next.js Example

Create a simple blog page:

```typescript
// app/page.tsx
import { SimplistClient } from '@simplist.blog/sdk';

const client = new SimplistClient();

export default async function HomePage() {
  const { data: articles } = await client.articles.list({
    published: true,
    limit: 10,
  });

  return (
    <div>
      <h1>Latest Articles</h1>
      <div className="grid gap-6">
        {articles.map((article) => (
          <article key={article.id} className="border p-6 rounded-lg">
            <h2 className="text-2xl font-bold">{article.title}</h2>
            <p className="text-gray-600 mt-2">{article.excerpt}</p>
            <div className="mt-4 flex gap-4 text-sm text-gray-500">
              <span>{article.readTimeMinutes} min read</span>
              <span>{article.viewCount} views</span>
            </div>
            <a
              href={`/articles/${article.slug}`}
              className="mt-4 inline-block text-blue-600 hover:underline"
            >
              Read more →
            </a>
          </article>
        ))}
      </div>
    </div>
  );
}
```

## React Hook Example

Create a custom hook for fetching articles:

```typescript
// hooks/useArticles.ts
import { SimplistClient } from '@simplist.blog/sdk';
import { useState, useEffect } from 'react';
import type { Article } from '@simplist.blog/sdk';

const client = new SimplistClient();

export function useArticles(limit: number = 10) {
  const [articles, setArticles] = useState<Article[]>([]);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState<Error | null>(null);

  useEffect(() => {
    async function fetchArticles() {
      try {
        setLoading(true);
        const { data } = await client.articles.list({ limit });
        setArticles(data);
      } catch (err) {
        setError(err instanceof Error ? err : new Error('Unknown error'));
      } finally {
        setLoading(false);
      }
    }

    fetchArticles();
  }, [limit]);

  return { articles, loading, error };
}

// Usage in component
function ArticleList() {
  const { articles, loading, error } = useArticles(10);

  if (loading) return <div>Loading...</div>;
  if (error) return <div>Error: {error.message}</div>;

  return (
    <div>
      {articles.map(article => (
        <div key={article.id}>{article.title}</div>
      ))}
    </div>
  );
}
```

## Error Handling

Always handle errors gracefully:

```typescript
import { SimplistClient, SimplistApiError } from '@simplist.blog/sdk';

const client = new SimplistClient();

try {
  const { data: article } = await client.articles.get('my-slug');
  console.log(article.title);
} catch (error) {
  if (error instanceof SimplistApiError) {
    // API error
    console.error(`API Error (${error.status}):`, error.message);

    if (error.status === 404) {
      console.log('Article not found');
    } else if (error.status === 401) {
      console.log('Invalid API key');
    } else if (error.status === 429) {
      console.log('Rate limit exceeded');
    }
  } else {
    // Network or other error
    console.error('Unexpected error:', error);
  }
}
```

## Common Patterns

### Fetch and Display Articles

```typescript
const { data: articles } = await client.articles.list({
  published: true,
  sort: 'publishedAt',
  order: 'desc',
  limit: 10,
});

articles.forEach(article => {
  console.log(article.title);
});
```

### Search Articles

```typescript
const { data: results } = await client.articles.search('typescript');

console.log(`Found ${results.length} articles about TypeScript`);
```

### Get Popular Articles

```typescript
const { data: popular } = await client.articles.list({
  published: true,
  sort: 'viewCount',
  order: 'desc',
  limit: 5,
});

console.log('Most popular articles:');
popular.forEach(article => {
  console.log(`${article.title} - ${article.viewCount} views`);
});
```

### Multi-Language Support

```typescript
const { data: article } = await client.articles.get('my-article');

// Check for French variant
const frenchVariant = article.variants?.find(v => v.lang === 'fr');

if (frenchVariant) {
  console.log('French title:', frenchVariant.title);
  console.log('French content:', frenchVariant.content);
}
```
