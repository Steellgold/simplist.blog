# Multilingual API

Simplist provides comprehensive multilingual support with **80+ languages**, automatic browser language detection, and smart content variant management.

## Supported Languages

Simplist supports 80+ languages including:

- **Popular**: English, French, Spanish, German, Italian, Portuguese, Russian
- **Asian**: Japanese, Korean, Chinese (Simplified/Traditional), Thai, Vietnamese
- **Middle Eastern**: Arabic, Hebrew, Persian
- **And many more**...

See the complete list in the [Language enum](#language-enum).

## VariantSelector Class

The `VariantSelector` class provides a consistent way to manage article variants across your application.

<MethodSignature
  name="new VariantSelector"
  parameters={[
    { name: "options", type: "{ defaultLanguage?: LanguageCode, userLanguage?: LanguageCode }", optional: true }
  ]}
  returnType="VariantSelector"
/>

### Methods

<ApiMethodTable
  methods={[
    { method: "getContent", description: "Get article content in preferred language", returns: "string" },
    { method: "getTitle", description: "Get article title in preferred language", returns: "string" },
    { method: "getExcerpt", description: "Get article excerpt in preferred language", returns: "string | null" },
    { method: "getFullContent", description: "Get complete article variant", returns: "ArticleVariant | Article" },
    { method: "getCoverImage", description: "Get cover image for variant", returns: "string | null" },
    { method: "getSelectedLanguage", description: "Get the selected language for article", returns: "LanguageCode" },
    { method: "setUserLanguage", description: "Update user language preference", returns: "void" },
    { method: "setDefaultLanguage", description: "Update default fallback language", returns: "void" }
  ]}
/>

### Example

<CodeBlock language="tsx">
{`
  import { VariantSelector } from '@simplist.blog/sdk'

  const selector = new VariantSelector({
    defaultLanguage: 'en',
    userLanguage: 'fr'
  })

  // Get localized content
  const title = selector.getTitle(article)
  const content = selector.getContent(article)
  const excerpt = selector.getExcerpt(article)

  // Get full variant
  const variant = selector.getFullContent(article)
  console.log(variant.lang)        // 'fr'
  console.log(variant.wordCount)   // Word count for French version

  // Change language dynamically
  selector.setUserLanguage('es')
  const spanishTitle = selector.getTitle(article)
`}
</CodeBlock>

## Complete Example

### Language Switcher Component

<CodeBlock language="tsx">
{`
  'use client'

  import { getAllLanguages, VariantSelector } from '@simplist.blog/sdk'
  import { useState } from 'react'
  import type { Article } from '@simplist.blog/sdk'

  export function LanguageSwitcher({ article }: { article: Article }) {
    const availableLanguages = getAllLanguages(article)
    const [selectedLang, setSelectedLang] = useState('en')

    const selector = new VariantSelector({
      defaultLanguage: 'en',
      userLanguage: selectedLang
    })

    const handleLanguageChange = (lang: string) => {
      setSelectedLang(lang)
      selector.setUserLanguage(lang as any)
    }

    return (
      <div>
        <select value={selectedLang} onChange={(e) => handleLanguageChange(e.target.value)}>
          {availableLanguages.map(lang => (
            <option key={lang} value={lang}>
              {lang.toUpperCase()}
            </option>
          ))}
        </select>

        <article>
          <h1>{selector.getTitle(article)}</h1>
          <p>{selector.getExcerpt(article)}</p>
          <div dangerouslySetInnerHTML={{ __html: selector.getContent(article) }} />
        </article>
      </div>
    )
  }
`}
</CodeBlock>

### Server Component with Auto-Detection

<CodeBlock language="tsx">
{`
  import { SimplistClient, getBestMatchingVariant } from '@simplist.blog/sdk'
  import { headers } from 'next/headers'

  export default async function ArticlePage({
    params
  }: {
    params: { slug: string }
  }) {
    // Uses SIMPLIST_API_KEY from environment variables
    const client = new SimplistClient()
    const { data: article } = await client.articles.get(params.slug)

    // Get user's language from Accept-Language header
    const headersList = headers()
    const acceptLanguage = headersList.get('accept-language')
    const userLang = acceptLanguage?.split(',')[0]?.split('-')[0] || 'en'

    // Get best matching variant
    const variant = getBestMatchingVariant(article)

    return (
      <article lang={variant.lang}>
        <h1>{variant.title}</h1>
        <p>{variant.excerpt}</p>
        <div dangerouslySetInnerHTML={{ __html: variant.content }} />

        {/* Show available translations */}
        <div className="translations">
          <h3>Available in:</h3>
          {getAllLanguages(article).map(lang => (
            <a key={lang} href={\`/\${lang}/\${article.slug}\`}>
              {lang.toUpperCase()}
            </a>
          ))}
        </div>
      </article>
    )
  }
`}
</CodeBlock>

## SEO Considerations

- **hreflang tags**: Add alternate language links for SEO
- **Language attribute**: Set `<html lang="...">` for each variant
- **Sitemap**: Include all language variants in sitemap
- **URL structure**: Use language prefix (e.g., `/fr/article-slug`)

<CodeBlock language="typescript">
{`
  // Generate hreflang tags
  import { getAllLanguages } from '@simplist.blog/sdk'

  export function generateHreflangTags(article: Article, baseUrl: string) {
    const languages = getAllLanguages(article)

    return languages.map(lang => ({
      rel: 'alternate',
      hrefLang: lang,
      href: \`\${baseUrl}/\${lang}/\${article.slug}\`
    }))
  }
`}
</CodeBlock>

## Best Practices

- **Fallback language**: Always set a default/fallback language (usually English)
- **User preference**: Remember user's language choice in localStorage/cookies
- **Auto-detection**: Use browser language for initial selection
- **Translation indicators**: Show which languages are available
