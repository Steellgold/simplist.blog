# Articles API

The Articles API provides methods to fetch, filter, search, and list articles from your Simplist project.

## list()

Retrieve a paginated list of articles with filtering and sorting options.

<MethodSignature
  name="client.articles.list"
  parameters={[
    { name: "params", type: "ArticleListParams", optional: true }
  ]}
  returnType="Promise<ApiResponse<ArticleListItem[]>>"
  async={true}
/>

### Parameters

<TypeTable
  title="ArticleListParams"
  properties={[
    { name: "page", type: "number", description: "Page number", defaultValue: "1" },
    { name: "limit", type: "number", description: "Items per page (max 100)", defaultValue: "10" },
    { name: "sort", type: "'createdAt' | 'updatedAt' | 'title'", description: "Sort field", defaultValue: "createdAt" },
    { name: "order", type: "'asc' | 'desc'", description: "Sort direction", defaultValue: "desc" },
    { name: "published", type: "boolean", description: "Filter by published status" },
    { name: "search", type: "string", description: "Search in title and content" },
    { name: "status", type: "'draft' | 'published'", description: "Filter by status" },
    { name: "optionalFields", type: "ArticleOptionalFields", description: "Request optional tag fields (color, icon)" }
  ]}
/>

### Example

<CodeBlock language="tsx">
{`

  // Basic list
  const { data, meta } = await client.articles.list({
    page: 1,
    limit: 20,
    sort: 'createdAt',
    order: 'desc',
    published: true
  })

  console.log(\`Found \${data.length} articles\`)
  console.log(\`Total: \${meta.total} (Page \${meta.page}/\${meta.totalPages})\`)

  data.forEach(article => {
    console.log(\`- \${article.title} (\${article.readTimeMinutes}min read)\`)
    console.log(\`  Tags: \${article.tags.map(t => t.name).join(', ')}\`)
  })

  // With optional tag fields (color and icon)
  const { data: articlesWithTags } = await client.articles.list({
    optionalFields: {
      tagColor: true,  // Include hex color codes
      tagIcon: true    // Include icon names
    }
  })

  articlesWithTags.forEach(article => {
    article.tags.forEach(tag => {
      console.log(\`Tag: \${tag.name}, Color: \${tag.color}, Icon: \${tag.icon}\`)
    })
  })
`}
</CodeBlock>

## Complete Example

### Blog List Page

<CodeBlock language="tsx">
{`
  import { SimplistClient } from '@simplist.blog/sdk'

  export default async function BlogPage({
    searchParams
  }: {
    searchParams: { page?: string }
  }) {
    // Uses SIMPLIST_API_KEY from environment variables
    const client = new SimplistClient()

    const page = Number(searchParams.page) || 1

    const { data: articles, meta } = await client.articles.published({
      page,
      limit: 10,
      sort: 'publishedAt',
      order: 'desc'
    })

    return (
      <div>
        <h1>Blog</h1>

        <div className="articles">
          {articles.map(article => (
            <article key={article.id}>
              <h2>{article.title}</h2>
              <p>{article.excerpt}</p>
              <span>{article.readTimeMinutes} min read</span>
              <span>{article.viewCount} views</span>
              <a href={\`/blog/\${article.slug}\`}>Read more</a>
            </article>
          ))}
        </div>

        <Pagination
          currentPage={meta.page}
          totalPages={meta.totalPages}
        />
      </div>
    )
  }
`}
</CodeBlock>

### Article Detail Page

<CodeBlock language="tsx">
{`
  import { SimplistClient } from '@simplist.blog/sdk'
  import { notFound } from 'next/navigation'

  export default async function ArticlePage({
    params
  }: {
    params: { slug: string }
  }) {
    // Uses SIMPLIST_API_KEY from environment variables
    const client = new SimplistClient()

    try {
      const { data: article } = await client.articles.get(params.slug)

      return (
        <article>
          <header>
            <h1>{article.title}</h1>
            <p>By {article.author.name}</p>
            <time>{new Date(article.publishedAt).toLocaleDateString()}</time>
            <span>{article.readTimeMinutes} min read</span>
          </header>

          <div dangerouslySetInnerHTML={{ __html: article.content }} />

          <footer>
            <p>{article.viewCount} views</p>
          </footer>
        </article>
      )
    } catch (error) {
      notFound()
    }
  }
`}
</CodeBlock>
