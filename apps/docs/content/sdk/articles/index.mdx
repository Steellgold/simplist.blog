# Articles Resource

The Articles resource provides methods to fetch and manage your blog articles.

## Available Methods

```typescript
import { SimplistClient } from '@simplist.blog/sdk';

const client = new SimplistClient();

// List articles
const { data, meta } = await client.articles.list({ limit: 10 });

// Get single article
const { data: article } = await client.articles.get('article-slug');

// Search articles
const { data: results } = await client.articles.search('typescript');

// Get published articles only
const { data: published } = await client.articles.published({ limit: 20 });

// Get latest articles
const { data: latest } = await client.articles.latest(5);

// Get popular articles
const { data: popular } = await client.articles.popular(10);

// Generate RSS feed
const rss = await client.articles.rss({
  hostname: 'https://myblog.com',
  title: 'My Blog',
  limit: 50,
});
```

## list()

Fetch articles with pagination and filtering.

```typescript
async list(params?: ArticleListParams): Promise<ApiResponse<ArticleListItem[]>>
```

### Parameters

```typescript
interface ArticleListParams {
  page?: number;        // Page number (default: 1)
  limit?: number;       // Items per page (default: 20, max: 100)
  sort?: string;        // Sort field (default: 'createdAt')
  order?: 'asc' | 'desc'; // Sort order (default: 'desc')
  published?: boolean;  // Filter by published status
  search?: string;      // Search in title, excerpt, content
  status?: 'draft' | 'published';
}
```

### Example

## get()

Fetch a single article by slug with full content and variants.

```typescript
async get(slug: string): Promise<ApiResponse<Article>>
```

### Example

```typescript
const { data: article } = await client.articles.get('getting-started');

console.log(article.title);
console.log(article.content); // Full Markdown content
console.log(`Reading time: ${article.readTimeMinutes} minutes`);

// Check for variants
if (article.variants && article.variants.length > 0) {
  const french = article.variants.find(v => v.lang === 'fr');
  if (french) {
    console.log('French title:', french.title);
  }
}
```

### Response

```typescript
{
  data: {
    id: string;
    title: string;
    slug: string;
    content: string;
    excerpt: string | null;
    coverImage: string | null;
    published: boolean;
    status: string;
    viewCount: number;
    readTimeMinutes: number;
    variants?: ArticleVariant[];
    // ... more fields
  }
}
```

## search()

Search articles by query string.

```typescript
async search(query: string, params?: Omit<ArticleListParams, 'search'>): Promise<ApiResponse<ArticleListItem[]>>
```

### Example

```typescript
const { data: results } = await client.articles.search('typescript', {
  published: true,
  limit: 20,
});

console.log(`Found ${results.length} articles about TypeScript`);
```

The search looks in:
- Article title
- Article excerpt
- Article content

<Callout type="info">
Search is case-insensitive and performs partial matching.
</Callout>

## published()

Shortcut to get only published articles.

```typescript
async published(params?: Omit<ArticleListParams, 'published'>): Promise<ApiResponse<ArticleListItem[]>>
```

### Example

```typescript
const { data: articles } = await client.articles.published({
  limit: 10,
  sort: 'publishedAt',
  order: 'desc',
});
```

Equivalent to:
```typescript
const { data: articles } = await client.articles.list({
  published: true,
  limit: 10,

  order: 'desc',

```

## latest()

Get the latest published articles.

```typescript
async latest(limit: number = 10): Promise<ApiResponse<ArticleListItem[]>>
```

### Example

```typescript
const { data: latest } = await client.articles.latest(5);

console.log('Latest 5 articles:');

  console.log(`- ${article.title}`);
});
```

Sorted by `createdAt` descending, published only.

## popular()


## rss()

Generate RSS feed XML for your articles.

```typescript
async rss(options: {
  hostname: string;
  title?: string;
  description?: string;
  limit?: number;
}): Promise<string>
```

### Example

```typescript
const rssXml = await client.articles.rss({
  hostname: 'https://myblog.com',
  title: 'My Blog',
  description: 'Latest articles from my blog',
  limit: 50,
});

```

### Next.js Route Example

```typescript
// app/rss.xml/route.ts
import { SimplistClient } from '@simplist.blog/sdk';
import { NextResponse } from 'next/server';

const client = new SimplistClient();

export async function GET() {
  const rss = await client.articles.rss({
    hostname: process.env.NEXT_PUBLIC_BASE_URL!,
    title: 'My Blog',
    limit: 50,
  });

  return new NextResponse(rss, {
    headers: {
      'Content-Type': 'application/rss+xml; charset=utf-8',
    },
  });
}
```

## Common Patterns

### Build Homepage Feed

```typescript
const { data: articles } = await client.articles.published({
  limit: 10,
  sort: 'publishedAt',
  order: 'desc',
});
```

### Search with Pagination

```typescript
async function searchWithPagination(query: string, page: number = 1) {
  return await client.articles.search(query, {
    page,
    limit: 20,
    published: true,
  });
}
```

### Get Article with Variant

```typescript
async function getArticleContent(slug: string, lang: string = 'en') {
  const { data: article } = await client.articles.get(slug);

  if (lang !== 'en' && article.variants) {
    const variant = article.variants.find(v => v.lang === lang);
    if (variant) {
      return {
        title: variant.title,
        content: variant.content,
        excerpt: variant.excerpt,
      };
    }
  }

  return {
    title: article.title,
    content: article.content,
    excerpt: article.excerpt,
  };
}
```

### List All Articles (Pagination)

```typescript
async function getAllArticles() {
  const allArticles: ArticleListItem[] = [];
  let page = 1;
  let hasMore = true;

  while (hasMore) {
    const { data, meta } = await client.articles.list({
      page,
      limit: 100,
      published: true,
    });

    allArticles.push(...data);
    hasMore = page < meta.totalPages;
    page++;
  }

  return allArticles;
}
```

## Error Handling

```typescript
import { SimplistApiError } from '@simplist.blog/sdk';

try {
  const { data: article } = await client.articles.get('my-slug');
} catch (error) {
  if (error instanceof SimplistApiError) {
    if (error.status === 404) {
      console.log('Article not found');
    } else if (error.status === 401) {
      console.log('Invalid API key');
    }
  }
}
```
