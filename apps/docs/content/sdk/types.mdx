## Usage Example

<CodeBlock language="typescript">
{`
  import type {
    SimplistClient,
    Article,
    ArticleListParams,
    ApiResponse,
    PageViewData,
    SeoMetadata
  } from '@simplist.blog/sdk'

  // Type-safe parameters
  const params: ArticleListParams = {
    page: 1,
    limit: 10,
    sort: 'createdAt',
    order: 'desc',
    published: true
  }

  // Type-safe response
  const response: ApiResponse<Article[]> = await client.articles.list(params)

  // Access with full type safety
  response.data.forEach((article: Article) => {
    console.log(article.title)
    console.log(article.readTimeMinutes)
  })
`}
</CodeBlock>
