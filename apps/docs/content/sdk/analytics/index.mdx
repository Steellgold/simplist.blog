# Analytics Resource

The Analytics resource provides methods to track page views and retrieve analytics statistics.

## Available Methods

```typescript
import { SimplistClient } from '@simplist.blog/sdk';

const client = new SimplistClient();

// Track page view
const result = await client.analytics.track({
  articleSlug: 'my-article',
  visitorId: 'visitor_123',
  sessionId: 'session_456',
  pageUrl: 'https://myblog.com/articles/my-article',
});

// Update page view metrics
await client.analytics.update(result.pageViewId, {
  timeOnPage: 180,
  scrollDepth: 75,
});

// Get statistics
const stats = await client.analytics.getStats({ days: 30 });
```

<Callout type="info">
Analytics tracking requires an API key with **analytics** permission. Statistics require **read** permission.
</Callout>

## track()

Create a new page view record.

```typescript
async track(data: PageViewData): Promise<PageViewResponse>
```

### Parameters

```typescript
interface PageViewData {
  // Required
  articleSlug: string;

  // Visitor tracking
  visitorId?: string;
  sessionId?: string;

  // Page info
  pageUrl?: string;
  pageTitle?: string;
  referrer?: string;

  // Device info
  screenWidth?: number;
  screenHeight?: number;

  // Geographic
  country?: string;
  countryCode?: string;
  region?: string;
  city?: string;
  timezone?: string;

  // UTM parameters
  utmSource?: string;
  utmMedium?: string;
  utmCampaign?: string;
  utmTerm?: string;
  utmContent?: string;

  // Engagement
  timeOnPage?: number;
  scrollDepth?: number;
  exitPosition?: number;
  bounced?: boolean;

  // Events
  events?: PageEvent[];
}
```

## update()

Update engagement metrics for an existing page view.

```typescript
async update(pageViewId: string, data: {
  timeOnPage?: number;
  scrollDepth?: number;
  exitPosition?: number;
  bounced?: boolean;
}): Promise<{ success: boolean }>
```

### Example

```typescript
// Track when user leaves the page
window.addEventListener('beforeunload', async () => {
  const pageViewId = sessionStorage.getItem('pageViewId');
  if (!pageViewId) return;

  const timeOnPage = Math.floor((Date.now() - pageLoadTime) / 1000);

  // Use sendBeacon for reliable tracking on page unload
  const data = JSON.stringify({
    timeOnPage,
    scrollDepth: maxScrollDepth,
    exitPosition: window.scrollY,
    bounced: false,
  });

  navigator.sendBeacon(
    `https://api.simplist.blog/v1/analytics/track/${pageViewId}`,
    new Blob([data], { type: 'application/json' })
  );
});
```

<Callout type="info">
Use `navigator.sendBeacon()` for reliable tracking when the page is unloading.
</Callout>

## getStats()

Retrieve aggregated analytics statistics.

```typescript
async getStats(options?: { days?: number }): Promise<AnalyticsStats>
```

### Example

```typescript
const stats = await client.analytics.getStats({ days: 30 });

console.log('Analytics Summary:');
console.log(`- Total views: ${stats.summary.totalViews}`);
console.log(`- Unique visitors: ${stats.summary.uniqueVisitors}`);
console.log(`- Avg views/visitor: ${stats.summary.avgViewsPerVisitor}`);

console.log('\nTop Articles:');
stats.topArticles.forEach((article, i) => {
  console.log(`${i + 1}. ${article.title} - ${article.views} views`);
});

console.log('\nTop Countries:');
stats.topCountries.forEach(country => {
  console.log(`- ${country.country}: ${country.views} views`);
});
```

### Response

```typescript
{
  period: {
    days: number;
    startDate: string;
    endDate: string;
  };
  summary: {
    totalViews: number;
    uniqueVisitors: number;
    avgViewsPerVisitor: number;
  };
  requestSource: {
    sdk: { count: number; percentage: number };
    direct: { count: number; percentage: number };
  };
  topArticles: Array<{
    articleId: string;
    title: string;
    slug: string;
    views: number;
  }>;
  topCountries: Array<{
    country: string;
    views: number;
  }>;
}
```

## Complete Tracking Example

```typescript
class PageAnalytics {
  private client: SimplistClient;
  private pageViewId: string | null = null;
  private startTime: number = Date.now();
  private maxScrollDepth: number = 0;

  constructor(apiKey: string) {
    this.client = new SimplistClient({ apiKey });
  }

  async trackPageLoad(articleSlug: string) {
    const result = await this.client.analytics.track({
      articleSlug,
      visitorId: this.getVisitorId(),
      sessionId: this.getSessionId(),
      pageUrl: window.location.href,
      referrer: document.referrer,
      screenWidth: window.screen.width,
      screenHeight: window.screen.height,
    });

    this.pageViewId = result.pageViewId;

    // Track scroll depth
    window.addEventListener('scroll', () => this.updateScrollDepth());

    // Update metrics on page unload
    window.addEventListener('beforeunload', () => this.trackPageExit());
  }

  private updateScrollDepth() {
    const scrollPercent = Math.min(
      100,
      Math.round(
        ((window.scrollY + window.innerHeight) / document.body.scrollHeight) * 100
      )
    );

    this.maxScrollDepth = Math.max(this.maxScrollDepth, scrollPercent);
  }

  private async trackPageExit() {
    if (!this.pageViewId) return;

    const timeOnPage = Math.floor((Date.now() - this.startTime) / 1000);
    const bounced = this.maxScrollDepth < 25 && timeOnPage < 10;

    await this.client.analytics.update(this.pageViewId, {
      timeOnPage,
      scrollDepth: this.maxScrollDepth,
      exitPosition: window.scrollY,
      bounced,
    });
  }

  private getVisitorId(): string {
    let visitorId = localStorage.getItem('visitorId');
    if (!visitorId) {
      visitorId = `visitor_${Date.now()}_${crypto.randomUUID()}`;
      localStorage.setItem('visitorId', visitorId);
    }
    return visitorId;
  }

  private getSessionId(): string {
    let sessionId = sessionStorage.getItem('sessionId');
    if (!sessionId) {
      sessionId = `session_${Date.now()}_${crypto.randomUUID()}`;
      sessionStorage.setItem('sessionId', sessionId);
    }
    return sessionId;
  }
}

// Usage
const analytics = new PageAnalytics(process.env.NEXT_PUBLIC_SIMPLIST_API_KEY!);
analytics.trackPageLoad('my-article-slug');
```

## React Hook Example

```typescript
import { SimplistClient } from '@simplist.blog/sdk';
import { useEffect } from 'react';

const client = new SimplistClient();

export function useArticleAnalytics(articleSlug: string) {
  useEffect(() => {
    let pageViewId: string | null = null;
    const startTime = Date.now();
    let maxScrollDepth = 0;

    async function trackPageView() {
      const result = await client.analytics.track({
        articleSlug,
        visitorId: getVisitorId(),
        sessionId: getSessionId(),
        pageUrl: window.location.href,
      });

      pageViewId = result.pageViewId;
    }

    function updateScrollDepth() {
      const scrollPercent = Math.min(
        100,
        Math.round(
          ((window.scrollY + window.innerHeight) / document.body.scrollHeight) * 100
        )
      );
      maxScrollDepth = Math.max(maxScrollDepth, scrollPercent);
    }

    async function trackExit() {
      if (!pageViewId) return;

      await client.analytics.update(pageViewId, {
        timeOnPage: Math.floor((Date.now() - startTime) / 1000),
        scrollDepth: maxScrollDepth,
      });
    }

    trackPageView();
    window.addEventListener('scroll', updateScrollDepth);
    window.addEventListener('beforeunload', trackExit);

    return () => {
      window.removeEventListener('scroll', updateScrollDepth);
      window.removeEventListener('beforeunload', trackExit);
      trackExit();
    };
  }, [articleSlug]);
}

// Usage in component
function ArticlePage({ slug }: { slug: string }) {
  useArticleAnalytics(slug);

  return <div>Article content...</div>;
}
```

## Bot Detection

The API automatically detects and blocks bot traffic. Requests from bots will return a `400 Bad Request` error.

<Callout type="info">
Bot detection is handled server-side. You don't need to implement it in your tracking code.
</Callout>

## Privacy

Analytics are privacy-first:
- IP addresses are **never stored**
- Visitor IDs are **project-specific**
- Geographic data is **optional**
- No cross-site tracking
